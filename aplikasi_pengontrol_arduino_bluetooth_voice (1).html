Kode Arduino C++ (.ino)

Unggah kode di bawah ke Arduino Anda. Kode ini mendukung perintah dari tombol digital, dimmer PWM, dan perintah suara. 

#include <SoftwareSerial.h>

SoftwareSerial BT(2, 3); // RX = 2, TX = 3 (Pin Modul Bluetooth)

const int RELAY_1 = 8;
const int RELAY_2 = 9;
const int RELAY_3 = 10;
const int RELAY_4 = 11;
const int PWM_PIN = 6; // Pin LED/Motor Dimmer

void setup() {
  Serial.begin(9600);
  BT.begin(9600); // Baudrate HC-05 / HC-06
  
  pinMode(RELAY_1, OUTPUT);
  pinMode(RELAY_2, OUTPUT);
  pinMode(RELAY_3, OUTPUT);
  pinMode(RELAY_4, OUTPUT);
  pinMode(PWM_PIN, OUTPUT);

  // Default High (Matikan relay jika relay Low Active)
  digitalWrite(RELAY_1, HIGH);
  digitalWrite(RELAY_2, HIGH);
  digitalWrite(RELAY_3, HIGH);
  digitalWrite(RELAY_4, HIGH);
}

void loop() {
  if (BT.available()) {
    String data = BT.readStringUntil('\n');
    data.trim(); // Hapus whitespace / newline
    
    Serial.println("Diterima: " + data);

    // Perintah Utama (Tombol & Suara)
    if (data == "A") digitalWrite(RELAY_1, LOW);  // Lampu 1 ON
    if (data == "a") digitalWrite(RELAY_1, HIGH); // Lampu 1 OFF
    if (data == "B") digitalWrite(RELAY_2, LOW);  // Lampu 2 ON
    if (data == "b") digitalWrite(RELAY_2, HIGH); // Lampu 2 OFF
    if (data == "C") digitalWrite(RELAY_3, LOW);  // Kipas/Relay 3 ON
    if (data == "c") digitalWrite(RELAY_3, HIGH); // Kipas/Relay 3 OFF
    if (data == "D") digitalWrite(RELAY_4, LOW);  // Stopkontak ON
    if (data == "d") digitalWrite(RELAY_4, HIGH); // Stopkontak OFF

    // Perintah Suara Semua / Power
    if (data == "ALL_ON" || data == "1") {
      digitalWrite(RELAY_1, LOW);
      digitalWrite(RELAY_2, LOW);
      digitalWrite(RELAY_3, LOW);
      digitalWrite(RELAY_4, LOW);
    }

    if (data == "ALL_OFF" || data == "0") {
      digitalWrite(RELAY_1, HIGH);
      digitalWrite(RELAY_2, HIGH);
      digitalWrite(RELAY_3, HIGH);
      digitalWrite(RELAY_4, HIGH);
    }

    // Handle Slider Dimmer (Format: "S:128")
    if (data.startsWith("S:")) {
      int pwmVal = data.substring(2).toInt();
      analogWrite(PWM_PIN, pwmVal);
    }
  }
}
